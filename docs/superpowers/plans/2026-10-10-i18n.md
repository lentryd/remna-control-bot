# i18n (en base + ru) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every bot and WebApp text exists in English (base, fallback) and Russian, chosen by the Telegram `language_code`, with friendlier wording and no internal jargon.

**Architecture:** A new Go package `internal/i18n` embeds flat JSON catalogs and hands out a `Localizer`. A bot middleware records the sender's language in SQLite and puts a `Localizer` into the request context; proactive messages read the stored language. The WebApp keeps i18next, flips its base to `en`, adds `ru`, detects the language from Telegram and moves the operator screens into an `admin` namespace.

**Tech Stack:** Go 1.26 (`embed`, `encoding/json`), go-telegram/bot, modernc sqlite, React 19 + i18next + react-i18next + http-backend, bun, TypeScript.

**Spec:** `docs/superpowers/specs/2026-10-10-i18n-design.md`. One deviation from the spec: the bot has no users table (customers live in remna-control), so the language is stored in a new table `user_lang(tg_id, lang)` instead of a `lang` column.

## Global Constraints

- Languages: `en` (base and fallback) and `ru`. A new language must need only new catalog files, no code changes.
- Language code normalization: primary subtag, lowercased (`ru-RU`, `ru_UA` → `ru`); unsupported → `en`.
- Missing key in a language → `en`; missing in `en` → the key itself.
- Catalog parity: every language has exactly the keys of `en` (plural suffixes aside) and the same format verbs per key.
- Wording: no internal jargon, short, warm, polite sentences. Replacements: «Админка» → «Управление» / "Manage"; «администратор» (to customers) → «поддержка» / "support"; «провайдер» → «платёжный сервис» / "payment service"; «панель» → «VPN-панель» / "VPN panel"; «квитанция» → «чек об оплате» / "payment receipt"; «аддоны» → «дополнения» / "extras". Payment-service names and their field names (Shop ID, API token, Crypto Pay) stay as the services call them.
- Telegram HTML parse mode is used by the bot: catalogs keep the existing `<b>`/`<code>` tags; arguments inserted into them are escaped by the caller exactly as today (`Esc`).
- Commits: no `Co-Authored-By` or any Claude attribution (user's CLAUDE.md overrides the harness reminder).
- Verification commands: `go test ./...` from the repo root; `cd web && bun run typecheck && bun run build`.

## Review Focus

1. Unknown or empty `language_code` (`""`, `"xx"`, `"pt-BR"`) → English, never a panic or empty text. Pinned in Task 1.
2. A new `ru` key forgotten or a `%s`/`%d` mismatch between languages → CI fails, not a runtime `%!s(MISSING)`. Pinned in Task 1 (Go) and Task 8 (web).
3. Russian plurals ("1 день / 2 дня / 5 дней / 11 дней / 21 день") are correct. Pinned in Task 1.
4. A proactive message (expiry notice, admin card) to a person whose language was never stored → English. Pinned in Task 2.
5. User changes the Telegram language → the next interaction updates the stored one. Pinned in Task 2.

```go
package i18n

// category returns the CLDR plural category used as a key suffix for n in lang.
// Unknown languages use the English rule. Add a case here when a language needs more forms.
func category(lang string, n int) string {
	if n < 0 {
		n = -n
	}
	switch lang {
	case "ru", "uk":
		switch {
		case n%10 == 1 && n%100 != 11:
			return "one"
		case n%10 >= 2 && n%10 <= 4 && (n%100 < 12 || n%100 > 14):
			return "few"
		default:
			return "many"
		}
	default:
		if n == 1 {
			return "one"
		}
		return "other"
	}
}
```

```go
// Package i18n holds the bot's text catalogs. Catalogs are flat JSON files
// (locales/<lang>.json) embedded into the binary; English is the base.
package i18n

import (
	"embed"
	"encoding/json"
	"fmt"
	"path"
	"sort"
	"strings"
)

//go:embed locales/*.json
var files embed.FS

// Base is the language every other one falls back to.
const Base = "en"

type Bundle struct{ cats map[string]map[string]string }

func Load() (*Bundle, error) {
	entries, err := files.ReadDir("locales")
	if err != nil {
		return nil, err
	}
	b := &Bundle{cats: map[string]map[string]string{}}
	for _, e := range entries {
		raw, err := files.ReadFile(path.Join("locales", e.Name()))
		if err != nil {
			return nil, err
		}
		cat := map[string]string{}
		if err := json.Unmarshal(raw, &cat); err != nil {
			return nil, fmt.Errorf("%s: %w", e.Name(), err)
		}
		b.cats[strings.TrimSuffix(e.Name(), ".json")] = cat
	}
	if b.cats[Base] == nil {
		return nil, fmt.Errorf("i18n: locales/%s.json is missing", Base)
	}
	return b, nil
}

// Default is the bundle built from the embedded catalogs.
var Default = func() *Bundle {
	b, err := Load()
	if err != nil {
		panic(err)
	}
	return b
}()

func (b *Bundle) Langs() []string {
	out := make([]string, 0, len(b.cats))
	for l := range b.cats {
		out = append(out, l)
	}
	sort.Strings(out)
	return out
}

// Resolve maps a Telegram language_code ("ru-RU", "pt_BR") to a supported language, or Base.
func (b *Bundle) Resolve(code string) string {
	code = strings.ToLower(strings.TrimSpace(code))
	if i := strings.IndexAny(code, "-_"); i >= 0 {
		code = code[:i]
	}
	if _, ok := b.cats[code]; ok {
		return code
	}
	return Base
}

func (b *Bundle) For(code string) Localizer { return Localizer{lang: b.Resolve(code), b: b} }

type Localizer struct {
	lang string
	b    *Bundle
}

func (l Localizer) bundle() *Bundle {
	if l.b == nil {
		return Default
	}
	return l.b
}

func (l Localizer) Lang() string {
	if l.lang == "" {
		return Base
	}
	return l.lang
}

func (l Localizer) lookup(key string) (string, bool) {
	b := l.bundle()
	if s, ok := b.cats[l.Lang()][key]; ok {
		return s, true
	}
	s, ok := b.cats[Base][key]
	return s, ok
}

func (l Localizer) Has(key string) bool { _, ok := l.lookup(key); return ok }

// T formats the catalog entry for key with fmt.Sprintf rules.
func (l Localizer) T(key string, args ...any) string {
	s, ok := l.lookup(key)
	if !ok {
		return key
	}
	if len(args) == 0 {
		return strings.ReplaceAll(s, "%%", "%")
	}
	return fmt.Sprintf(s, args...)
}

// Plural picks key_<category of n> (falling back to key_other) and formats it with args.
func (l Localizer) Plural(key string, n int, args ...any) string {
	if k := key + "_" + category(l.Lang(), n); l.Has(k) {
		return l.T(k, args...)
	}
	return l.T(key+"_other", args...)
}
```

```go
// SetLang remembers the Telegram language of a person; an empty lang changes nothing.
func (s *Store) SetLang(ctx context.Context, tgID int64, lang string) error {
	if lang == "" {
		return nil
	}
	_, err := s.db.ExecContext(ctx,
		`INSERT INTO user_lang(tg_id, lang) VALUES(?, ?) ON CONFLICT(tg_id) DO UPDATE SET lang = excluded.lang`, tgID, lang)
	return err
}

// Lang returns the remembered language, or "" when the person is unknown.
func (s *Store) Lang(ctx context.Context, tgID int64) (string, error) {
	var lang string
	err := s.db.QueryRowContext(ctx, `SELECT lang FROM user_lang WHERE tg_id = ?`, tgID).Scan(&lang)
	if errors.Is(err, sql.ErrNoRows) {
		return "", nil
	}
	return lang, err
}
```

```go
package telegram

import (
	"context"

	tgbot "github.com/go-telegram/bot"
	"github.com/go-telegram/bot/models"

	"remna-control-bot/internal/i18n"
)

// LangStore remembers each person's Telegram language for messages the bot sends on its own (*store.Store).
type LangStore interface {
	SetLang(ctx context.Context, tgID int64, lang string) error
	Lang(ctx context.Context, tgID int64) (string, error)
}

type locKey struct{}

// langFrom returns the sender of an update and the language_code Telegram reports for them.
func langFrom(u *models.Update) (tgID int64, code string, ok bool) {
	switch {
	case u.Message != nil && u.Message.From != nil:
		return u.Message.From.ID, u.Message.From.LanguageCode, true
	case u.CallbackQuery != nil:
		return u.CallbackQuery.From.ID, u.CallbackQuery.From.LanguageCode, true
	case u.PreCheckoutQuery != nil && u.PreCheckoutQuery.From != nil:
		return u.PreCheckoutQuery.From.ID, u.PreCheckoutQuery.From.LanguageCode, true
	}
	return 0, "", false
}

// langMiddleware stores the sender's language and puts their Localizer into the context.
func (b *Bot) langMiddleware() tgbot.Middleware {
	return func(next tgbot.HandlerFunc) tgbot.HandlerFunc {
		return func(ctx context.Context, tg *tgbot.Bot, u *models.Update) {
			if id, code, ok := langFrom(u); ok {
				loc := i18n.Default.For(code)
				if code == "" {
					loc = b.locFor(ctx, id)
				} else if b.Langs != nil {
					if err := b.Langs.SetLang(ctx, id, loc.Lang()); err != nil && b.Log != nil {
						b.Log.Warn("store language", "err", err)
					}
				}
				ctx = context.WithValue(ctx, locKey{}, loc)
			}
			next(ctx, tg, u)
		}
	}
}

// loc is the localizer of the person whose update is being handled (English outside a handler).
func (b *Bot) loc(ctx context.Context) i18n.Localizer {
	l, _ := ctx.Value(locKey{}).(i18n.Localizer)
	return l
}

// locFor is the localizer for a person the bot writes to unprompted; English when their language is unknown.
func (b *Bot) locFor(ctx context.Context, tgID int64) i18n.Localizer {
	if b.Langs == nil {
		return i18n.Default.For("")
	}
	code, err := b.Langs.Lang(ctx, tgID)
	if err != nil && b.Log != nil {
```

```ts
import { readdirSync, readFileSync, existsSync } from 'node:fs'
import { join } from 'node:path'

const root = join(import.meta.dir, '..', 'public', 'locales')
const flat = (o: unknown, p = ''): Record<string, string> =>
    Object.entries(o as Record<string, unknown>).reduce((acc, [k, v]) => {
        const key = p ? `${p}.${k}` : k
        return typeof v === 'object' && v !== null ? { ...acc, ...flat(v, key) } : { ...acc, [key]: String(v) }
    }, {} as Record<string, string>)
const vars = (s: string) => (s.match(/{{\s*\w+\s*}}/g) ?? []).map(x => x.replace(/\s/g, '')).sort().join(',')
const plural = /_(zero|one|two|few|many|other)$/
const base = (k: string) => k.replace(plural, '')

let bad = 0
const langs = readdirSync(root)
const namespaces = new Set(langs.flatMap(l => readdirSync(join(root, l))))
for (const ns of namespaces) {
    const enFile = join(root, 'en', ns)
    if (!existsSync(enFile)) { console.error(`en/${ns} is missing`); bad++; continue }
    const en = flat(JSON.parse(readFileSync(enFile, 'utf8')))
    const enBase = new Map(Object.entries(en).map(([k, v]) => [base(k), vars(v)]))
    for (const lang of langs.filter(l => l !== 'en')) {
        const file = join(root, lang, ns)
        if (!existsSync(file)) { console.error(`${lang}/${ns} is missing`); bad++; continue }
        const other = flat(JSON.parse(readFileSync(file, 'utf8')))
        const otherBase = new Map(Object.entries(other).map(([k, v]) => [base(k), vars(v)]))
        for (const [k, v] of enBase) {
            if (!otherBase.has(k)) { console.error(`${lang}/${ns}: missing ${k}`); bad++ }
            else if (otherBase.get(k) !== v) { console.error(`${lang}/${ns}: ${k} placeholders differ`); bad++ }
        }
        for (const k of otherBase.keys()) if (!enBase.has(k)) { console.error(`${lang}/${ns}: extra ${k}`); bad++ }
    }
}
process.exit(bad ? 1 : 0)
```
